//|                                                                  |
//| VERSION HISTORY                                                  |
//|   1.0.0  Ricochet / GoldRetraceTrap prototypes (superseded)      |
//|   1.0.1  Commercial rebuild: broker-state reconciler, net-of-    |
//|          cost basket TP, dashboard, minimal inputs               |
//|                                                                  |
//| ARCHITECTURE (read before changing anything)                     |
//|                                                                  |
//|   BROKER STATE  -> BuildSnapshot()      (positions, orders,      |
//|                                          deal history)           |
//|                 -> ManageBasket()/ManagePair()                   |
//|                    decides ONE action category per cycle         |
//|                 -> CTrade request                                |
//|                 -> result recorded only as an in-flight guard    |
//|                    or a cooldown                                 |
//|                 -> next OnTick/OnTimer re-reads BROKER STATE.    |
//|                                                                  |
//| The EA keeps no belief about what the broker "should" contain.   |
//| Local state is limited to: cooldowns, in-flight guards for       |
//| requests whose result is not yet visible, the fingerprint of the |
//| last basket whose TPs were synchronised, and an observed         |
//| commission model. Everything else is re-derived every cycle.     |
//|                                                                  |
//| STRATEGY (unchanged from the validated prototype)                |
//|   1. SELL LIMIT above Ask and BUY LIMIT below Bid (entry pair).  |
//|   2. One side fills -> that side is the basket, the opposite     |
//|      pending is cancelled.                                       |
//|   3. Basket carries a monetary target (net of costs).            |
//|   4. Adverse move >= ATR(M1,5) from the latest fill -> one       |
//|      same-direction recovery position, lot x 1.675 per level.    |
//|   5. Every basket position shares one TP derived from the        |
//|      volume-weighted average price.                              |
//|   6. Basket closes -> a new entry pair is built.                 |
//+------------------------------------------------------------------+

enum ENUM_RX_STATE
  {
   RX_STATE_STARTING,
   RX_STATE_BLOCKED,          // trading not possible right now (terminal / account / symbol)
   RX_STATE_NO_BASKET,        // nothing of ours on the broker
   RX_STATE_PAIR_PARTIAL,     // exactly one entry leg present
   RX_STATE_PAIR_ACTIVE,      // both entry legs present
   RX_STATE_BASKET_ACTIVE,    // one position
   RX_STATE_RECOVERY_ACTIVE,  // two or more positions
   RX_STATE_BASKET_CLOSING    // position count is shrinking
  };

// Everything the strategy needs, re-read from the broker every cycle.
struct BrokerSnapshot
  {
   int                positions;        // basket positions (basket side only)
   int                strayPositions;   // our positions on the opposite side (anomaly)
   ENUM_POSITION_TYPE side;             // side of the earliest position
   double             totalVolume;
   double             vwap;
   double             lastEntryPrice;   // most recent fill of the basket side
   long               lastEntryTimeMsc;
   ulong              lastEntryTicket;
   double             floatingProfit;   // profit + swap
   double             totalSwap;        // signed, negative = cost
   double             entryCommission;  // signed, from deals, negative = cost
   double             commonTP;         // TP shared by all basket positions
   bool               uniformTP;
   string             fingerprint;      // changes whenever basket composition changes
   int                sellLimits;
   int                buyLimits;
   ulong              sellLimitTicket;
   ulong              buyLimitTicket;
   double             sellLimitPrice;
   double             buyLimitPrice;
  };

int OnInit()
  {
   g_symbol      = _Symbol;
   g_startupTick = GetTickCount();
   g_uiPrefix    = StringFormat("%s_%I64u_", PRODUCT_NAME, InpMagicNumber);
   g_lockName    = StringFormat("%s.lock.%s.%I64u", PRODUCT_NAME, g_symbol, InpMagicNumber);

   if(InpInitialLot <= 0.0 || InpBasketProfitTarget <= 0.0 ||
      InpEntryDistance < 0.0 || InpMaxPositions < 0)
     {
      Alert(PRODUCT_NAME, ": invalid settings. Initial Lot and Basket Profit Target must be > 0, ",
            "Entry Distance and Max Basket Positions must be >= 0.");
      return(INIT_PARAMETERS_INCORRECT);
     }

   if(!SymbolInfoInteger(g_symbol, SYMBOL_SELECT))
      SymbolSelect(g_symbol, true);
   RefreshSymbolConstants();

   g_atrHandle = iATR(g_symbol, ATR_TIMEFRAME, ATR_PERIOD);
   if(g_atrHandle == INVALID_HANDLE)
     {
      Alert(PRODUCT_NAME, ": volatility indicator could not be created (error ", GetLastError(), ").");
      return(INIT_FAILED);
     }

   g_trade.SetExpertMagicNumber(InpMagicNumber);
   g_trade.SetTypeFilling(ChooseFillingMode());
   g_trade.SetDeviationInPoints(MIN_DEVIATION_POINTS);
   g_trade.SetAsyncMode(false);
   g_trade.LogLevel(LOG_LEVEL_NO);      // RecoilX logs its own outcomes

   g_wasConnected = (bool)TerminalInfoInteger(TERMINAL_CONNECTED);
   CheckInstanceLock();

   if(InpShowDashboard && !MQLInfoInteger(MQL_OPTIMIZATION))
      DashboardCreate();

   EventSetTimer(TIMER_SECONDS);

   Log(StringFormat("%s %s started | %s | lot=%s | target=%.2f %s net | entry distance=%s | magic=%I64u",
                    PRODUCT_NAME, PRODUCT_RELEASE, g_symbol,
                    DoubleToString(InpInitialLot, g_volumeDigits),
                    InpBasketProfitTarget, AccountInfoString(ACCOUNT_CURRENCY),
                    InpEntryDistance > 0.0 ? DoubleToString(InpEntryDistance, _Digits) : "auto",
                    InpMagicNumber));
   // No trade requests here: the broker state may not be synchronised yet.
   return(INIT_SUCCEEDED);
  }

void OnDeinit(const int reason)
  {
   EventKillTimer();
   if(g_atrHandle != INVALID_HANDLE)
     {
      IndicatorRelease(g_atrHandle);
      g_atrHandle = INVALID_HANDLE;
     }
   ObjectsDeleteAll(0, g_uiPrefix);
   ChartRedraw();
   if(!MQLInfoInteger(MQL_TESTER) && GlobalVariableCheck(g_lockName))
      GlobalVariableDel(g_lockName);
   // Pending orders and positions are intentionally left untouched: the next
   // start reconciles them from broker state.
  }

void OnTick()
  {
   TrackConnection();
   Reconcile();
  }

void OnTimer()
  {
   if(!MQLInfoInteger(MQL_TESTER))
      GlobalVariableSet(g_lockName, (double)TimeLocal());
   TrackConnection();
   Reconcile();
  }

// Observation only. Trading decisions are made by Reconcile() from broker state.
void OnTradeTransaction(const MqlTradeTransaction &trans,
                        const MqlTradeRequest &request,
                        const MqlTradeResult &result)
  {
   if(trans.type == TRADE_TRANSACTION_DEAL_ADD)
     {
      if(!HistoryDealSelect(trans.deal))
         return;
      if(HistoryDealGetString(trans.deal, DEAL_SYMBOL) != g_symbol)
         return;
      if((ulong)HistoryDealGetInteger(trans.deal, DEAL_MAGIC) != InpMagicNumber)
         return;

      ENUM_DEAL_ENTRY entry = (ENUM_DEAL_ENTRY)HistoryDealGetInteger(trans.deal, DEAL_ENTRY);
      ENUM_DEAL_TYPE  type  = (ENUM_DEAL_TYPE)HistoryDealGetInteger(trans.deal, DEAL_TYPE);
      double vol   = HistoryDealGetDouble(trans.deal, DEAL_VOLUME);
      double price = HistoryDealGetDouble(trans.deal, DEAL_PRICE);
      double comm  = HistoryDealGetDouble(trans.deal, DEAL_COMMISSION) + HistoryDealGetDouble(trans.deal, DEAL_FEE);
      ulong  order = (ulong)HistoryDealGetInteger(trans.deal, DEAL_ORDER);
      string dir   = (type == DEAL_TYPE_BUY) ? "BUY" : "SELL";

      if(entry == DEAL_ENTRY_IN)
        {
         if(g_inflightActive && order == g_inflightOrder)
            g_inflightActive = false;
         if(vol > 0.0)
            g_entryCommPerLot = MathMax(0.0, -comm / vol);
         LogEvent(StringFormat("Filled %s %s @ %s (commission %.2f)",
                               dir, DoubleToString(vol, g_volumeDigits),
                               DoubleToString(price, _Digits), comm));
        }
      else if(entry == DEAL_ENTRY_OUT || entry == DEAL_ENTRY_OUT_BY || entry == DEAL_ENTRY_INOUT)
        {
         if(vol > 0.0)
            g_exitCommPerLot = MathMax(0.0, -comm / vol);
         LogEvent(StringFormat("Closed %s @ %s | P/L %.2f | commission %.2f | swap %.2f",
                               DoubleToString(vol, g_volumeDigits), DoubleToString(price, _Digits),
                               HistoryDealGetDouble(trans.deal, DEAL_PROFIT), comm,
                               HistoryDealGetDouble(trans.deal, DEAL_SWAP)));
        }
      return;
     }

   if(trans.type == TRADE_TRANSACTION_HISTORY_ADD)
     {
      if(trans.symbol != g_symbol || !HistoryOrderSelect(trans.order))
         return;
      if((ulong)HistoryOrderGetInteger(trans.order, ORDER_MAGIC) != InpMagicNumber)
         return;
      ENUM_ORDER_STATE st = (ENUM_ORDER_STATE)HistoryOrderGetInteger(trans.order, ORDER_STATE);
      ENUM_ORDER_TYPE  ot = (ENUM_ORDER_TYPE)HistoryOrderGetInteger(trans.order, ORDER_TYPE);
      if((ot == ORDER_TYPE_BUY_LIMIT || ot == ORDER_TYPE_SELL_LIMIT) &&
         (st == ORDER_STATE_CANCELED || st == ORDER_STATE_EXPIRED || st == ORDER_STATE_REJECTED))
         LogEvent(StringFormat("%s #%I64u removed (%s)", OrderTypeName(ot), trans.order, OrderStateName(st)));
     }
  }

//+------------------------------------------------------------------+
//| Reconciler                                                       |
//+------------------------------------------------------------------+
void TrackConnection()
  {
   bool connected = (bool)TerminalInfoInteger(TERMINAL_CONNECTED);
   if(connected && !g_wasConnected)
     {
      g_nextPairAttempt     = 0;
      g_nextRecoveryAttempt = 0;
      g_nextTpSyncAttempt   = 0;
      LogEvent("Connection restored - reconciling with broker state");
     }
   g_wasConnected = connected;
  }

void Reconcile()
  {
   RefreshSymbolConstants();

   BrokerSnapshot s;
   BuildSnapshot(s);
   UpdateInflight(s);

   string blockReason;
   if(!EnvironmentReady(blockReason))
     {
      g_state      = RX_STATE_BLOCKED;
      g_statusText = blockReason;
     }
   else
     {
      if(!g_commissionLearned)
         LearnCommissionFromHistory();

      if(s.positions > 0)
         ManageBasket(s);
      else
         ManagePair(s);
     }

   g_lastPositions = s.positions;
   DashboardUpdate(s);
  }

bool EnvironmentReady(string &reason)
  {
   bool tester = (bool)MQLInfoInteger(MQL_TESTER);

   if(!tester && (GetTickCount() - g_startupTick) < STARTUP_GRACE_MS)
     { reason = "Starting"; return(false); }
   if(!tester && !TerminalInfoInteger(TERMINAL_CONNECTED))
     { reason = "No connection to trade server"; return(false); }
   if(!TerminalInfoInteger(TERMINAL_TRADE_ALLOWED))
     { reason = "Algo trading disabled in terminal"; return(false); }
   if(!MQLInfoInteger(MQL_TRADE_ALLOWED))
     { reason = "Algo trading disabled for this chart"; return(false); }
   if(!AccountInfoInteger(ACCOUNT_TRADE_ALLOWED))
     { reason = "Trading disabled on account"; return(false); }
   if(!AccountInfoInteger(ACCOUNT_TRADE_EXPERT))
     { reason = "Expert trading disabled on account"; return(false); }
   if((ENUM_ACCOUNT_MARGIN_MODE)AccountInfoInteger(ACCOUNT_MARGIN_MODE) != ACCOUNT_MARGIN_MODE_RETAIL_HEDGING)
     {
      if(!g_nettingAlerted)
        {
         Alert(PRODUCT_NAME, ": this account is not a hedging account. Trading is paused.");
         g_nettingAlerted = true;
        }
      reason = "Hedging account required";
      return(false);
     }
   if(SymbolInfoInteger(g_symbol, SYMBOL_TRADE_MODE) != SYMBOL_TRADE_MODE_FULL)
     { reason = "Symbol trading restricted by broker"; return(false); }
   if(!LicenseAllowsTrading())
     { reason = "License inactive"; return(false); }

   return(true);
  }

//+------------------------------------------------------------------+
//| Broker snapshot                                                  |
//+------------------------------------------------------------------+
void ResetSnapshot(BrokerSnapshot &s)
  {
   s.positions = 0;      s.strayPositions = 0;   s.side = POSITION_TYPE_BUY;
   s.totalVolume = 0.0;  s.vwap = 0.0;           s.lastEntryPrice = 0.0;
   s.lastEntryTimeMsc = 0; s.lastEntryTicket = 0;
   s.floatingProfit = 0.0; s.totalSwap = 0.0;    s.entryCommission = 0.0;
   s.commonTP = 0.0;     s.uniformTP = true;     s.fingerprint = "none";
   s.sellLimits = 0;     s.buyLimits = 0;
   s.sellLimitTicket = 0; s.buyLimitTicket = 0;
   s.sellLimitPrice = 0.0; s.buyLimitPrice = 0.0;
  }

bool IsOurSelectedPosition()
  {
   return(PositionGetString(POSITION_SYMBOL) == g_symbol &&
          (ulong)PositionGetInteger(POSITION_MAGIC) == InpMagicNumber);
  }

bool IsOurSelectedOrder()
  {
   return(OrderGetString(ORDER_SYMBOL) == g_symbol &&
          (ulong)OrderGetInteger(ORDER_MAGIC) == InpMagicNumber);
  }

// Only orders that are genuinely resting count as pending legs.
bool IsRestingOrderState(const ENUM_ORDER_STATE st)
  {
   return(st == ORDER_STATE_PLACED || st == ORDER_STATE_PARTIAL ||
          st == ORDER_STATE_STARTED || st == ORDER_STATE_REQUEST_ADD ||
          st == ORDER_STATE_REQUEST_MODIFY);
  }

void BuildSnapshot(BrokerSnapshot &s)
  {
   ResetSnapshot(s);

   // Pass 1: the basket side is the side of the EARLIEST position we own.
   bool  found = false;
   long  earliest = 0;
   ulong earliestTicket = 0;
   int   total = PositionsTotal();
   for(int i = 0; i < total; ++i)
     {
      ulong ticket = PositionGetTicket(i);        // selects the position
      if(ticket == 0 || !IsOurSelectedPosition())
         continue;
      long t = PositionGetInteger(POSITION_TIME_MSC);
      if(!found || t < earliest || (t == earliest && ticket < earliestTicket))
        {
         found = true;
         earliest = t;
         earliestTicket = ticket;
         s.side = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
        }
     }

   if(found)
     {
      // Pass 2: aggregate the basket side.
      double weighted = 0.0;
      ulong  ticketSum = 0, ticketXor = 0;
      for(int i = 0; i < total; ++i)
        {
         ulong ticket = PositionGetTicket(i);
         if(ticket == 0 || !IsOurSelectedPosition())
            continue;
         if((ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE) != s.side)
           {
            s.strayPositions++;
            continue;
           }
         double v     = PositionGetDouble(POSITION_VOLUME);
         double open  = PositionGetDouble(POSITION_PRICE_OPEN);
         double tp    = PositionGetDouble(POSITION_TP);
         double swap  = PositionGetDouble(POSITION_SWAP);
         double prof  = PositionGetDouble(POSITION_PROFIT);
         long   t     = PositionGetInteger(POSITION_TIME_MSC);
         long   posId = PositionGetInteger(POSITION_IDENTIFIER);

         s.positions++;
         s.totalVolume    += v;
         weighted         += open * v;
         s.floatingProfit += prof + swap;
         s.totalSwap      += swap;
         if(t > s.lastEntryTimeMsc || (t == s.lastEntryTimeMsc && ticket > s.lastEntryTicket))
           {
            s.lastEntryTimeMsc = t;
            s.lastEntryTicket  = ticket;
            s.lastEntryPrice   = open;
           }
         if(s.positions == 1)
            s.commonTP = tp;
         else if(MathAbs(tp - s.commonTP) > g_tickSize * 0.5)
            s.uniformTP = false;
         ticketSum += ticket;
         ticketXor ^= ticket;

         // Deal history lookup (cached per ticket). Position fields were read above,
         // so the history selection below cannot disturb them.
         s.entryCommission += PositionEntryCommission(ticket, posId, v);
        }
      if(s.totalVolume > 0.0)
         s.vwap = weighted / s.totalVolume;

      s.fingerprint = StringFormat("%d|%s|%I64u|%I64u|%.2f|%s",
                                   s.positions, DoubleToString(s.totalVolume, 8),
                                   ticketSum, ticketXor, s.totalSwap,
                                   DoubleToString(s.uniformTP ? s.commonTP : -1.0, _Digits));
     }

   // Pending entry legs.
   int orders = OrdersTotal();
   for(int i = 0; i < orders; ++i)
     {
      ulong ticket = OrderGetTicket(i);          // selects the order
      if(ticket == 0 || !IsOurSelectedOrder())
         continue;
      if(!IsRestingOrderState((ENUM_ORDER_STATE)OrderGetInteger(ORDER_STATE)))
         continue;
      ENUM_ORDER_TYPE type = (ENUM_ORDER_TYPE)OrderGetInteger(ORDER_TYPE);
      double price = OrderGetDouble(ORDER_PRICE_OPEN);
      if(type == ORDER_TYPE_SELL_LIMIT)
        {
         s.sellLimits++;
         s.sellLimitTicket = ticket;
         s.sellLimitPrice  = price;
        }
      else if(type == ORDER_TYPE_BUY_LIMIT)
        {
         s.buyLimits++;
         s.buyLimitTicket = ticket;
         s.buyLimitPrice  = price;
        }
     }
  }

// Clears the in-flight guard once the broker reflects the accepted request.
void UpdateInflight(const BrokerSnapshot &s)
  {
   if(!g_inflightActive)
      return;

   int total = PositionsTotal();
   for(int i = 0; i < total; ++i)
     {
      ulong ticket = PositionGetTicket(i);
      if(ticket == 0 || !IsOurSelectedPosition())
         continue;
      if((ulong)PositionGetInteger(POSITION_IDENTIFIER) == g_inflightOrder)
        {
         g_inflightActive = false;
         return;
        }
     }

   if(HistoryOrderSelect(g_inflightOrder))
     {
      ENUM_ORDER_STATE st = (ENUM_ORDER_STATE)HistoryOrderGetInteger(g_inflightOrder, ORDER_STATE);
      if(st == ORDER_STATE_REJECTED || st == ORDER_STATE_CANCELED || st == ORDER_STATE_EXPIRED)
        {
         g_inflightActive = false;
         LogEvent(StringFormat("Recovery order #%I64u ended as %s", g_inflightOrder, OrderStateName(st)));
         return;
        }
     }

   if(TimeCurrent() - g_inflightSince > INFLIGHT_TIMEOUT_SEC)
     {
      g_inflightActive = false;
      WarnThrottled("inflight", StringFormat("Order #%I64u accepted but never reflected in positions; continuing from broker state", g_inflightOrder));
     }
  }

// True when a deal opened one of our positions very recently but the position
// is not yet visible and has not been closed. This is the window in which the
// old implementation could rebuild or delete the entry pair incorrectly.
bool HasUnreflectedRecentEntry()
  {
   datetime now = TimeCurrent();
   if(!HistorySelect(now - INFLIGHT_TIMEOUT_SEC, now + 86400))
      return(false);

   long  inIds[];
   long  outIds[];
   int deals = HistoryDealsTotal();
   for(int i = 0; i < deals; ++i)
     {
      ulong deal = HistoryDealGetTicket(i);
      if(deal == 0)
         continue;
      if(HistoryDealGetString(deal, DEAL_SYMBOL) != g_symbol)
         continue;
      if((ulong)HistoryDealGetInteger(deal, DEAL_MAGIC) != InpMagicNumber)
         continue;
      long posId = HistoryDealGetInteger(deal, DEAL_POSITION_ID);
      ENUM_DEAL_ENTRY entry = (ENUM_DEAL_ENTRY)HistoryDealGetInteger(deal, DEAL_ENTRY);
      if(entry == DEAL_ENTRY_IN)
        {
         int n = ArraySize(inIds);
         ArrayResize(inIds, n + 1);
         inIds[n] = posId;
        }
      else
        {
         int n = ArraySize(outIds);
         ArrayResize(outIds, n + 1);
         outIds[n] = posId;
        }
     }

   for(int i = 0; i < ArraySize(inIds); ++i)
     {
      bool closed = false;
      for(int j = 0; j < ArraySize(outIds) && !closed; ++j)
         if(outIds[j] == inIds[i])
            closed = true;
      if(closed)
         continue;
      if(!PositionVisible(inIds[i]))
         return(true);
     }
   return(false);
  }

bool PositionVisible(const long positionId)
  {
   int total = PositionsTotal();
   for(int i = 0; i < total; ++i)
     {
      ulong ticket = PositionGetTicket(i);
      if(ticket == 0 || !IsOurSelectedPosition())
         continue;
      if(PositionGetInteger(POSITION_IDENTIFIER) == positionId)
         return(true);
     }
   return(false);
  }

//+------------------------------------------------------------------+
//| Basket management                                                |
//+------------------------------------------------------------------+
void ManageBasket(const BrokerSnapshot &s)
  {
   datetime now = TimeCurrent();

   if(g_lastPositions > s.positions && g_lastPositions > 0)
      g_shrinkSeenAt = now;
   bool settling = (g_shrinkSeenAt > 0 && (now - g_shrinkSeenAt) < BASKET_SETTLE_SEC);

   if(settling)
     { g_state = RX_STATE_BASKET_CLOSING;  g_statusText = "Basket closing"; }
   else if(s.positions == 1)
     { g_state = RX_STATE_BASKET_ACTIVE;   g_statusText = "Basket active - waiting for target"; }
   else
     { g_state = RX_STATE_RECOVERY_ACTIVE; g_statusText = StringFormat("Recovery active (%d positions)", s.positions); }

   if(s.strayPositions > 0)
      WarnThrottled("stray", StringFormat("%d opposite-side position(s) exist alongside the %s basket; they keep their own TP and are excluded from the basket",
                    s.strayPositions, s.side == POSITION_TYPE_BUY ? "BUY" : "SELL"));

   // 1. A basket exists: only the OPPOSITE entry leg is removed. Nothing else.
   ENUM_ORDER_TYPE opposite = (s.side == POSITION_TYPE_BUY) ? ORDER_TYPE_SELL_LIMIT : ORDER_TYPE_BUY_LIMIT;
   int oppositeCount = (s.side == POSITION_TYPE_BUY) ? s.sellLimits : s.buyLimits;
   if(oppositeCount > 0)
     {
      DeleteOurPendingOrders(opposite, "opposite entry leg of active basket");
      return;
     }

   if(settling)
      return;

   // 2. Common TP, only when the basket composition changed since the last sync.
   if(s.fingerprint != g_syncedFingerprint && now >= g_nextTpSyncAttempt)
     {
      int rc = SyncBasketTP(s);
      if(rc == 1)
         g_syncedFingerprint = s.fingerprint;
      else if(rc == 2)
         return;                                   // basket is being closed at market
      else if(rc < 0)
         g_nextTpSyncAttempt = now + TP_SYNC_RETRY_SEC;
      else
         g_nextTpSyncAttempt = now + 1;            // deferred (stops/freeze zone)
     }

   // 3. Recovery: exactly one per adverse level.
   if(InpMaxPositions > 0 && s.positions >= InpMaxPositions)
     {
      g_nextRecoveryPrice = 0.0;
      return;
     }

   double required = 0.0;
   bool triggered = RecoveryTriggerReached(s, required);
   g_nextRecoveryPrice = (required > 0.0)
                         ? (s.side == POSITION_TYPE_BUY ? s.lastEntryPrice - required : s.lastEntryPrice + required)
                         : 0.0;
   if(!triggered)
      return;
   if(g_inflightActive || now < g_nextRecoveryAttempt)
     {
      if(now < g_nextRecoveryAttempt)
         g_statusText = "Recovery paused - retrying shortly";
      return;
     }
   if(HasUnreflectedRecentEntry())
      return;

   OpenRecovery(s);
  }

bool RecoveryTriggerReached(const BrokerSnapshot &s, double &required)
  {
   required = 0.0;
   double atr = CurrentATR();
   g_lastATR = atr;
   if(atr <= 0.0 || s.lastEntryPrice <= 0.0)
      return(false);
   required = atr * RECOVERY_ATR_MULTIPLIER;

   MqlTick tick;
   if(!SymbolInfoTick(g_symbol, tick))
      return(false);

   if(s.side == POSITION_TYPE_BUY)
      return((s.lastEntryPrice - tick.bid) >= required);   // BUY basket loses as price falls
   return((tick.ask - s.lastEntryPrice) >= required);      // SELL basket loses as price rises
  }

void OpenRecovery(const BrokerSnapshot &s)
  {
   datetime now = TimeCurrent();
   double lot = NextLot(s.positions);

   MqlTick tick;
   if(!SymbolInfoTick(g_symbol, tick))
      return;

   ENUM_ORDER_TYPE type = (s.side == POSITION_TYPE_BUY) ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
   double price = (s.side == POSITION_TYPE_BUY) ? tick.ask : tick.bid;

   // Local margin pre-check: avoids sending a request the broker will reject.
   double requiredMargin = 0.0;
   if(OrderCalcMargin(type, g_symbol, lot, price, requiredMargin) &&
      requiredMargin > AccountInfoDouble(ACCOUNT_MARGIN_FREE))
     {
      ReportRecoveryBlocked(lot, requiredMargin, "insufficient free margin");
      g_nextRecoveryAttempt = now + RESOURCE_BACKOFF_SEC;
      return;
     }

   g_trade.SetDeviationInPoints(DeviationPoints(tick));
   bool ok = (s.side == POSITION_TYPE_BUY)
             ? g_trade.Buy(lot, g_symbol, 0.0, 0.0, 0.0, ORDER_COMMENT)
             : g_trade.Sell(lot, g_symbol, 0.0, 0.0, 0.0, ORDER_COMMENT);

   if(ok)
     {
      g_inflightActive = true;
      g_inflightOrder  = g_trade.ResultOrder();
      g_inflightSince  = now;
      LogEvent(StringFormat("Recovery %s %s sent (level %d) @ %s",
                            s.side == POSITION_TYPE_BUY ? "BUY" : "SELL",
                            DoubleToString(lot, g_volumeDigits), s.positions + 1,
                            DoubleToString(g_trade.ResultPrice(), _Digits)));
      return;
     }

   uint rc = g_trade.ResultRetcode();
   if(rc == TRADE_RETCODE_NO_MONEY || rc == TRADE_RETCODE_LIMIT_VOLUME ||
      rc == TRADE_RETCODE_LIMIT_POSITIONS || rc == TRADE_RETCODE_LIMIT_ORDERS)
     {
      ReportRecoveryBlocked(lot, requiredMargin, g_trade.ResultRetcodeDescription());
      g_nextRecoveryAttempt = now + RESOURCE_BACKOFF_SEC;
     }
   else if(rc == TRADE_RETCODE_MARKET_CLOSED || rc == TRADE_RETCODE_TRADE_DISABLED)
     {
      LogEvent(StringFormat("Recovery not possible: %s", g_trade.ResultRetcodeDescription()));
      g_nextRecoveryAttempt = now + MARKET_CLOSED_RETRY_SEC;
     }
   else
     {
      LogEvent(StringFormat("Recovery request failed: %s (%u)", g_trade.ResultRetcodeDescription(), rc));
      g_nextRecoveryAttempt = now + RECOVERY_RETRY_SEC;
     }
  }

void ReportRecoveryBlocked(const double lot, const double requiredMargin, const string reason)
  {
   LogEvent(StringFormat("RECOVERY BLOCKED | lot %s | %s | free margin %.2f | required %.2f | pausing %ds",
                         DoubleToString(lot, g_volumeDigits), reason,
                         AccountInfoDouble(ACCOUNT_MARGIN_FREE), requiredMargin, RESOURCE_BACKOFF_SEC));
  }

// Returns 1 = all TPs match target, 0 = deferred (price too close), -1 = request failed,
// 2 = target already exceeded so the basket was closed at market.
int SyncBasketTP(const BrokerSnapshot &s)
  {
   double costs = BasketCosts(s);
   double gross = InpBasketProfitTarget + costs;
   double tp    = TargetPriceForGross(s.side, s.vwap, s.totalVolume, gross);
   if(tp <= 0.0)
     {
      WarnThrottled("tpcalc", "Basket TP could not be calculated (profit calculation unavailable for this symbol)");
      return(-1);
     }
   g_lastTP = tp;
   g_lastGrossTarget = gross;
   g_lastCosts = costs;

   MqlTick tick;
   if(!SymbolInfoTick(g_symbol, tick))
      return(-1);

   double minDist = MinStopDistance();
   bool exceeded = (s.side == POSITION_TYPE_BUY) ? (tp <= tick.bid) : (tp >= tick.ask);
   if(exceeded)
     {
      LogEvent("Target already reached but TP cannot be placed behind price - closing basket at market");
      CloseBasketAtMarket(s, tick);
      return(2);
     }
   bool insideStops = (s.side == POSITION_TYPE_BUY) ? ((tp - tick.bid) < minDist) : ((tick.ask - tp) < minDist);
   if(insideStops)
      return(0);   // price is about to reach the target; the broker rejects TPs this close

   // Collect basket tickets, newest first (preserves the prototype's modification order).
   ulong tickets[];
   long  times[];
   int total = PositionsTotal();
   for(int i = 0; i < total; ++i)
     {
      ulong ticket = PositionGetTicket(i);
      if(ticket == 0 || !IsOurSelectedPosition())
         continue;
      if((ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE) != s.side)
         continue;
      int n = ArraySize(tickets);
      ArrayResize(tickets, n + 1);
      ArrayResize(times, n + 1);
      tickets[n] = ticket;
      times[n]   = PositionGetInteger(POSITION_TIME_MSC);
     }
   for(int a = 1; a < ArraySize(tickets); ++a)
     {
      ulong tk = tickets[a];
      long  tm = times[a];
      int b = a - 1;
      while(b >= 0 && times[b] < tm)
        {
         tickets[b + 1] = tickets[b];
         times[b + 1]   = times[b];
         --b;
        }
      tickets[b + 1] = tk;
      times[b + 1]   = tm;
     }

   double freezeDist = SymbolInfoInteger(g_symbol, SYMBOL_TRADE_FREEZE_LEVEL) * _Point;
   int modified = 0, failed = 0, deferred = 0;
   for(int j = 0; j < ArraySize(tickets); ++j)
     {
      if(!PositionSelectByTicket(tickets[j]))
         continue;
      double curTP = PositionGetDouble(POSITION_TP);
      double curSL = PositionGetDouble(POSITION_SL);
      if(MathAbs(curTP - tp) <= g_tickSize * 0.5)
         continue;
      if(curTP > 0.0 && freezeDist > 0.0)
        {
         double ref = (s.side == POSITION_TYPE_BUY) ? tick.bid : tick.ask;
         if(MathAbs(curTP - ref) < freezeDist)
           {
            deferred++;
            continue;
           }
        }
      if(g_trade.PositionModify(tickets[j], curSL, tp))
         modified++;
      else
        {
         failed++;
         Log(StringFormat("TP modify failed #%I64u: %s", tickets[j], g_trade.ResultRetcodeDescription()));
        }
     }

   if(modified > 0)
      LogEvent(StringFormat("Basket TP %s | %s x%d | avg %s | gross %.2f (net %.2f + costs %.2f)",
                            DoubleToString(tp, _Digits), s.side == POSITION_TYPE_BUY ? "BUY" : "SELL",
                            s.positions, DoubleToString(s.vwap, _Digits),
                            gross, InpBasketProfitTarget, costs));
   if(failed > 0)
      return(-1);
   if(deferred > 0)
      return(0);
   return(1);
  }

void CloseBasketAtMarket(const BrokerSnapshot &s, const MqlTick &tick)
  {
   g_trade.SetDeviationInPoints(DeviationPoints(tick));
   int total = PositionsTotal();
   for(int i = total - 1; i >= 0; --i)
     {
      ulong ticket = PositionGetTicket(i);
      if(ticket == 0 || !IsOurSelectedPosition())
         continue;
      if((ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE) != s.side)
         continue;
      if(!g_trade.PositionClose(ticket))
         Log(StringFormat("Close failed #%I64u: %s", ticket, g_trade.ResultRetcodeDescription()));
     }
   g_nextTpSyncAttempt = TimeCurrent() + TP_SYNC_RETRY_SEC;
  }

//+------------------------------------------------------------------+
//| Entry pair management (only runs when we hold NO positions)      |
//+------------------------------------------------------------------+
void ManagePair(const BrokerSnapshot &s)
  {
   datetime now = TimeCurrent();

   if(g_lastPositions > 0)
      OnBasketClosed();

   bool sellTracked = LegAwaitingVisibility(0);
   bool buyTracked  = LegAwaitingVisibility(1);
   bool hasSell = (s.sellLimits > 0) || sellTracked;
   bool hasBuy  = (s.buyLimits  > 0) || buyTracked;

   if(hasSell && hasBuy)
     { g_state = RX_STATE_PAIR_ACTIVE;  g_statusText = "Waiting for entry"; }
   else if(hasSell || hasBuy)
     { g_state = RX_STATE_PAIR_PARTIAL; g_statusText = "Entry pair incomplete - repairing"; }
   else
     { g_state = RX_STATE_NO_BASKET;    g_statusText = "Placing entry orders"; }

   // Steady state: both legs present, no duplicates. Nothing to do.
   if(hasSell && hasBuy && s.sellLimits <= 1 && s.buyLimits <= 1)
      return;

   if(now < g_nextPairAttempt)
      return;

   if(!TradingSessionOpen())
     {
      g_statusText = "Market closed";
      return;
     }

   // Duplicate legs (two instances, or a placement that became visible late):
   // keep the one nearest the market, remove the rest.
   if(s.sellLimits > 1)
     { DeleteDuplicateLegs(ORDER_TYPE_SELL_LIMIT); return; }
   if(s.buyLimits > 1)
     { DeleteDuplicateLegs(ORDER_TYPE_BUY_LIMIT); return; }

   // A fill may be materialising into a position we cannot see yet. Touching
   // the pair now is exactly how both legs used to disappear. Wait.
   if(g_inflightActive || HasUnreflectedRecentEntry())
     {
      g_statusText = "Entry filled - waiting for position";
      return;
     }

   double dist = EntryDistance();
   if(dist <= 0.0)
     {
      WarnThrottled("entrydist", "Entry distance unavailable (volatility data not ready)");
      return;
     }

   // Lone leg far from the market: rebuild around the current price.
   if(hasSell != hasBuy)
     {
      ENUM_ORDER_TYPE lone = hasSell ? ORDER_TYPE_SELL_LIMIT : ORDER_TYPE_BUY_LIMIT;
      double lonePrice = hasSell ? s.sellLimitPrice : s.buyLimitPrice;
      if(lonePrice > 0.0 && IsLegStale(lone, lonePrice, dist))
        {
         DeleteOurPendingOrders(lone, "stale lone entry leg");
         g_nextPairAttempt = now + 1;
         return;
        }
     }

   bool ok = true;
   if(!hasSell)
      ok = PlacePendingLeg(ORDER_TYPE_SELL_LIMIT, dist) && ok;
   if(!hasBuy)
      ok = PlacePendingLeg(ORDER_TYPE_BUY_LIMIT, dist) && ok;

   if(!ok)
     {
      uint rc = g_trade.ResultRetcode();
      bool closed = (rc == TRADE_RETCODE_MARKET_CLOSED || rc == TRADE_RETCODE_TRADE_DISABLED);
      g_nextPairAttempt = now + (closed ? MARKET_CLOSED_RETRY_SEC : PAIR_RETRY_SEC);
     }
  }

void OnBasketClosed()
  {
   g_syncedFingerprint = "";
   g_shrinkSeenAt      = 0;
   g_lastTP            = 0.0;
   g_lastGrossTarget   = 0.0;
   g_lastCosts         = 0.0;
   g_nextRecoveryPrice = 0.0;
   g_nextTpSyncAttempt = 0;
   ArrayResize(g_commCacheTicket, 0);
   ArrayResize(g_commCacheValue, 0);
   LogEvent("Basket closed - starting a new cycle");
  }

// True while a leg we placed has been accepted but is not yet visible in OrdersTotal().
bool LegAwaitingVisibility(const int idx)
  {
   if(g_legTicket[idx] == 0)
      return(false);
   if(OrderSelect(g_legTicket[idx]))
     {
      g_legTicket[idx] = 0;            // visible: the snapshot counts it from now on
      return(false);
     }
   if(HistoryOrderSelect(g_legTicket[idx]))
     {
      g_legTicket[idx] = 0;            // filled / cancelled / rejected: no longer pending
      return(false);
     }
   if(TimeCurrent() - g_legPlacedAt[idx] > INFLIGHT_TIMEOUT_SEC)
     {
      g_legTicket[idx] = 0;
      return(false);
     }
   return(true);
  }

bool PlacePendingLeg(const ENUM_ORDER_TYPE type, double dist)
  {
   MqlTick tick;
   if(!SymbolInfoTick(g_symbol, tick))
      return(false);

   double lot     = NormalizeVolume(InpInitialLot);
   double minDist = MinStopDistance() + g_tickSize;
   if(dist < minDist)
     {
      WarnThrottled("mindist", StringFormat("Entry distance %s raised to broker minimum %s",
                    DoubleToString(dist, _Digits), DoubleToString(minDist, _Digits)));
      dist = minDist;
     }

   ENUM_POSITION_TYPE side = (type == ORDER_TYPE_SELL_LIMIT) ? POSITION_TYPE_SELL : POSITION_TYPE_BUY;
   double price = (type == ORDER_TYPE_SELL_LIMIT) ? NormalizePrice(tick.ask + dist)
                                                  : NormalizePrice(tick.bid - dist);

   // Provisional TP from the order price; re-synchronised from the real fill and
   // real entry commission as soon as the position is visible.
   double gross = InpBasketProfitTarget + EstimatedRoundTurnCommission(lot);
   double tp    = TargetPriceForGross(side, price, lot, gross);
   if(tp > 0.0 && MathAbs(price - tp) < minDist)
      tp = NormalizePrice(side == POSITION_TYPE_BUY ? price + minDist : price - minDist);

   bool ok = (type == ORDER_TYPE_SELL_LIMIT)
             ? g_trade.SellLimit(lot, price, g_symbol, 0.0, tp, ORDER_TIME_GTC, 0, ORDER_COMMENT)
             : g_trade.BuyLimit(lot, price, g_symbol, 0.0, tp, ORDER_TIME_GTC, 0, ORDER_COMMENT);

   int idx = (type == ORDER_TYPE_SELL_LIMIT) ? 0 : 1;
   if(ok)
     {
      g_legTicket[idx]   = g_trade.ResultOrder();
      g_legPlacedAt[idx] = TimeCurrent();
      LogEvent(StringFormat("%s placed @ %s | lot %s | TP %s", OrderTypeName(type),
                            DoubleToString(price, _Digits), DoubleToString(lot, g_volumeDigits),
                            DoubleToString(tp, _Digits)));
     }
   else
      LogEvent(StringFormat("%s rejected: %s (%u)", OrderTypeName(type),
                            g_trade.ResultRetcodeDescription(), g_trade.ResultRetcode()));
   return(ok);
  }

bool IsLegStale(const ENUM_ORDER_TYPE type, const double legPrice, const double dist)
  {
   MqlTick tick;
   if(!SymbolInfoTick(g_symbol, tick))
      return(false);
   double reference = (type == ORDER_TYPE_SELL_LIMIT) ? tick.ask : tick.bid;
   return(MathAbs(legPrice - reference) >= dist * STALE_LEG_MULTIPLE);
  }

// Deletes our resting orders of exactly one type on this symbol.
void DeleteOurPendingOrders(const ENUM_ORDER_TYPE type, const string why)
  {
   int orders = OrdersTotal();
   for(int i = orders - 1; i >= 0; --i)
     {
      ulong ticket = OrderGetTicket(i);
      if(ticket == 0 || !IsOurSelectedOrder())
         continue;
      if((ENUM_ORDER_TYPE)OrderGetInteger(ORDER_TYPE) != type)
         continue;
      if(!IsRestingOrderState((ENUM_ORDER_STATE)OrderGetInteger(ORDER_STATE)))
         continue;
      if(g_trade.OrderDelete(ticket))
         LogEvent(StringFormat("%s #%I64u cancelled (%s)", OrderTypeName(type), ticket, why));
      else
         Log(StringFormat("Cancel failed #%I64u: %s", ticket, g_trade.ResultRetcodeDescription()));
     }
  }

void DeleteDuplicateLegs(const ENUM_ORDER_TYPE type)
  {
   MqlTick tick;
   if(!SymbolInfoTick(g_symbol, tick))
      return;
   double reference = (type == ORDER_TYPE_SELL_LIMIT) ? tick.ask : tick.bid;

   ulong keep = 0;
   double keepDist = 0.0;
   int orders = OrdersTotal();
   for(int i = 0; i < orders; ++i)
     {
      ulong ticket = OrderGetTicket(i);
      if(ticket == 0 || !IsOurSelectedOrder())
         continue;
      if((ENUM_ORDER_TYPE)OrderGetInteger(ORDER_TYPE) != type)
         continue;
      double d = MathAbs(OrderGetDouble(ORDER_PRICE_OPEN) - reference);
      if(keep == 0 || d < keepDist)
        {
         keep = ticket;
         keepDist = d;
        }
     }
   for(int i = orders - 1; i >= 0; --i)
     {
      ulong ticket = OrderGetTicket(i);
      if(ticket == 0 || ticket == keep || !IsOurSelectedOrder())
         continue;
      if((ENUM_ORDER_TYPE)OrderGetInteger(ORDER_TYPE) != type)
         continue;
      if(g_trade.OrderDelete(ticket))
         LogEvent(StringFormat("Duplicate %s #%I64u removed", OrderTypeName(type), ticket));
      else
         Log(StringFormat("Duplicate cancel failed #%I64u: %s", ticket, g_trade.ResultRetcodeDescription()));
     }
  }

//+------------------------------------------------------------------+
//| Money target -> price                                            |
//+------------------------------------------------------------------+
// Account-currency profit per 1.0 price unit for this volume, using the
// broker's own calculation (handles contract size, tick value and profit
// currency conversion). A wide probe avoids the 2-decimal rounding of
// OrderCalcProfit on low-value symbols such as BTC with small lots.
double ProfitPerPriceUnit(const ENUM_POSITION_TYPE side, const double volume, const double refPrice)
  {
   ENUM_ORDER_TYPE type = (side == POSITION_TYPE_BUY) ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
   double probe = MathMax(g_tickSize * 1000.0, refPrice * 0.01);
   double close = (side == POSITION_TYPE_BUY) ? refPrice + probe : refPrice - probe;
   double profit = 0.0;
   if(OrderCalcProfit(type, g_symbol, volume, refPrice, close, profit) && profit > 0.0)
      return(profit / probe);

   double tickValue = SymbolInfoDouble(g_symbol, SYMBOL_TRADE_TICK_VALUE_PROFIT);
   if(tickValue <= 0.0)
      tickValue = SymbolInfoDouble(g_symbol, SYMBOL_TRADE_TICK_VALUE);
   if(tickValue > 0.0 && g_tickSize > 0.0)
      return(volume * tickValue / g_tickSize);
   return(0.0);
  }

// Price at which closing `volume` opened at `refPrice` yields at least `grossTarget`.
double TargetPriceForGross(const ENUM_POSITION_TYPE side, const double refPrice,
                           const double volume, const double grossTarget)
  {
   if(refPrice <= 0.0 || volume <= 0.0 || g_tickSize <= 0.0)
      return(0.0);
   double perUnit = ProfitPerPriceUnit(side, volume, refPrice);
   if(perUnit <= 0.0)
      return(0.0);

   double distance = MathMax(grossTarget, 0.0) / perUnit;
   double ticks = MathCeil(distance / g_tickSize - 1e-9);   // round AWAY from entry
   if(ticks < 1.0)
      ticks = 1.0;
   double tp = (side == POSITION_TYPE_BUY) ? refPrice + ticks * g_tickSize : refPrice - ticks * g_tickSize;
   tp = NormalizeDouble(tp, _Digits);

   // Confirm with the broker's calculation; nudge outward if rounding fell short.
   ENUM_ORDER_TYPE type = (side == POSITION_TYPE_BUY) ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
   for(int i = 0; i < 5; ++i)
     {
      double profit = 0.0;
      if(!OrderCalcProfit(type, g_symbol, volume, refPrice, tp, profit))
         break;
      if(profit >= grossTarget - 0.005)
         break;
      tp = NormalizeDouble((side == POSITION_TYPE_BUY) ? tp + g_tickSize : tp - g_tickSize, _Digits);
     }
   return(tp);
  }

//+------------------------------------------------------------------+
//| Cost model (commission + swap), observed from the broker         |
//+------------------------------------------------------------------+
double ExitCommissionPerLot()
  {
   return(g_exitCommPerLot >= 0.0 ? g_exitCommPerLot : g_entryCommPerLot);
  }

double EstimatedRoundTurnCommission(const double volume)
  {
   return((g_entryCommPerLot + ExitCommissionPerLot()) * volume);
  }

// Positive = cost the TP must recover on top of the net target.
double BasketCosts(const BrokerSnapshot &s)
  {
   double entry = -s.entryCommission;                      // deals store costs as negatives
   double exit  = ExitCommissionPerLot() * s.totalVolume;   // estimate
   double swap  = -s.totalSwap;                             // negative swap is a cost
   return(entry + exit + swap);
  }

// Signed entry commission (+fees) of one position from its IN deals. Cached per ticket.
double PositionEntryCommission(const ulong ticket, const long positionId, const double volume)
  {
   int n = ArraySize(g_commCacheTicket);
   for(int i = 0; i < n; ++i)
      if(g_commCacheTicket[i] == ticket)
         return(g_commCacheValue[i]);

   double comm = 0.0, vol = 0.0;
   bool any = false;
   if(HistorySelectByPosition(positionId))
     {
      int deals = HistoryDealsTotal();
      for(int d = 0; d < deals; ++d)
        {
         ulong deal = HistoryDealGetTicket(d);
         if(deal == 0)
            continue;
         if((ENUM_DEAL_ENTRY)HistoryDealGetInteger(deal, DEAL_ENTRY) != DEAL_ENTRY_IN)
            continue;
         comm += HistoryDealGetDouble(deal, DEAL_COMMISSION) + HistoryDealGetDouble(deal, DEAL_FEE);
         vol  += HistoryDealGetDouble(deal, DEAL_VOLUME);
         any = true;
        }
     }
   if(!any)
      return(-g_entryCommPerLot * volume);   // history not available yet: estimate, do not cache

   if(vol > 0.0)
      g_entryCommPerLot = MathMax(0.0, -comm / vol);

   ArrayResize(g_commCacheTicket, n + 1);
   ArrayResize(g_commCacheValue, n + 1);
   g_commCacheTicket[n] = ticket;
   g_commCacheValue[n]  = comm;
   return(comm);
  }

// Seeds the commission model from recent closed trades of this EA on this symbol.
void LearnCommissionFromHistory()
  {
   datetime now = TimeCurrent();
   if(now == 0)
      return;
   if(!HistorySelect(now - COMMISSION_HISTORY_DAYS * 86400, now + 86400))
      return;
   g_commissionLearned = true;

   bool inSeen = false, outSeen = false;
   int deals = HistoryDealsTotal();
   for(int i = deals - 1; i >= 0 && !(inSeen && outSeen); --i)
     {
      ulong deal = HistoryDealGetTicket(i);
      if(deal == 0)
         continue;
      if(HistoryDealGetString(deal, DEAL_SYMBOL) != g_symbol)
         continue;
      if((ulong)HistoryDealGetInteger(deal, DEAL_MAGIC) != InpMagicNumber)
         continue;
      double vol = HistoryDealGetDouble(deal, DEAL_VOLUME);
      if(vol <= 0.0)
         continue;
      double comm = HistoryDealGetDouble(deal, DEAL_COMMISSION) + HistoryDealGetDouble(deal, DEAL_FEE);
      ENUM_DEAL_ENTRY entry = (ENUM_DEAL_ENTRY)HistoryDealGetInteger(deal, DEAL_ENTRY);
      if(entry == DEAL_ENTRY_IN && !inSeen)
        {
         g_entryCommPerLot = MathMax(0.0, -comm / vol);
         inSeen = true;
        }
      else if((entry == DEAL_ENTRY_OUT || entry == DEAL_ENTRY_OUT_BY) && !outSeen)
        {
         g_exitCommPerLot = MathMax(0.0, -comm / vol);
         outSeen = true;
        }
     }
   Log(StringFormat("Cost model: entry commission %.2f/lot (%s), exit commission %.2f/lot (%s)",
                    g_entryCommPerLot, inSeen ? "observed" : "none yet",
                    ExitCommissionPerLot(), outSeen ? "observed" : "assumed = entry"));
  }

void RefreshSymbolConstants()
  {
   g_tickSize = SymbolInfoDouble(g_symbol, SYMBOL_TRADE_TICK_SIZE);
   if(g_tickSize <= 0.0)
      g_tickSize = _Point;

   double step = SymbolInfoDouble(g_symbol, SYMBOL_VOLUME_STEP);
   int d = 0;
   if(step > 0.0)
      while(d < 8 && MathAbs(step - MathRound(step)) > 1e-12)
        {
         step *= 10.0;
         ++d;
        }
   else
      d = 2;
   g_volumeDigits = d;
  }

double EntryDistance()
  {
   if(InpEntryDistance > 0.0)
      return(InpEntryDistance);
   double atr = CurrentATR();
   return(atr > 0.0 ? atr * AUTO_ENTRY_ATR_MULTIPLIER : 0.0);
  }

double CurrentATR()
  {
   if(g_atrHandle == INVALID_HANDLE)
      return(0.0);
   double buffer[];                       // dynamic: ArraySetAsSeries is only valid here
   ArraySetAsSeries(buffer, true);
   if(CopyBuffer(g_atrHandle, 0, ATR_SHIFT, 1, buffer) != 1)
      return(0.0);
   return(buffer[0] > 0.0 ? buffer[0] : 0.0);
  }

double MinStopDistance()
  {
   long stops  = SymbolInfoInteger(g_symbol, SYMBOL_TRADE_STOPS_LEVEL);
   long freeze = SymbolInfoInteger(g_symbol, SYMBOL_TRADE_FREEZE_LEVEL);
   return(MathMax(stops, freeze) * _Point);
  }

int DeviationPoints(const MqlTick &tick)
  {
   double spreadPoints = (tick.ask - tick.bid) / _Point;
   return((int)MathMax(MIN_DEVIATION_POINTS, MathRound(spreadPoints * 3.0)));
  }

ENUM_ORDER_TYPE_FILLING ChooseFillingMode()
  {
   long modes = SymbolInfoInteger(g_symbol, SYMBOL_FILLING_MODE);
   if((modes & SYMBOL_FILLING_FOK) != 0)
      return(ORDER_FILLING_FOK);
   if((modes & SYMBOL_FILLING_IOC) != 0)
      return(ORDER_FILLING_IOC);
   return(ORDER_FILLING_RETURN);
  }

// Uses the broker's published trading sessions; if none are published the
// market is assumed open and the MARKET_CLOSED retcode backoff takes over.
bool TradingSessionOpen()
  {
   MqlDateTime dt;
   TimeToStruct(TimeTradeServer(), dt);
   long secondsOfDay = dt.hour * 3600 + dt.min * 60 + dt.sec;
   datetime from = 0, to = 0;
   int i = 0;
   for(; SymbolInfoSessionTrade(g_symbol, (ENUM_DAY_OF_WEEK)dt.day_of_week, i, from, to); ++i)
      if(secondsOfDay >= (long)from && secondsOfDay < (long)to)
         return(true);
   return(i == 0);
  }

double NextLot(const int currentPositions)
  {
   return(NormalizeVolume(InpInitialLot * MathPow(LOT_MULTIPLIER, currentPositions)));
  }

double NormalizePrice(const double price)
  {
   return(NormalizeDouble(MathRound(price / g_tickSize) * g_tickSize, _Digits));
  }

double NormalizeVolume(double volume)
  {
   double minV = SymbolInfoDouble(g_symbol, SYMBOL_VOLUME_MIN);
   double maxV = SymbolInfoDouble(g_symbol, SYMBOL_VOLUME_MAX);
   double step = SymbolInfoDouble(g_symbol, SYMBOL_VOLUME_STEP);
   if(step <= 0.0)
      step = 0.01;
   if(maxV <= 0.0)
      maxV = volume;
   volume = MathMax(minV, MathMin(maxV, volume));
   volume = MathRound(volume / step) * step;
   return(NormalizeDouble(volume, g_volumeDigits));
  }

// Warns (does not block) if another instance appears to manage the same symbol+magic.
void CheckInstanceLock()
  {
   if(MQLInfoInteger(MQL_TESTER))
      return;
   if(GlobalVariableCheck(g_lockName))
     {
      datetime beat = (datetime)GlobalVariableGet(g_lockName);
      if(TimeLocal() - beat < INSTANCE_LOCK_STALE_SEC)
         Alert(PRODUCT_NAME, ": another instance appears to run on ", g_symbol,
               " with magic ", InpMagicNumber, ". Use one chart per symbol or a different Magic Number.");
     }
   GlobalVariableSet(g_lockName, (double)TimeLocal());
  }

void WarnThrottled(const string key, const string msg)
  {
   datetime now = TimeCurrent();
   if(key == g_lastWarnKey && (now - g_lastWarnAt) < WARN_REPEAT_SEC)
      return;
   g_lastWarnKey = key;
   g_lastWarnAt  = now;
   Log("WARNING: " + msg);
  }

void DashboardUpdate(const BrokerSnapshot &s)
  {
   if(!InpShowDashboard || MQLInfoInteger(MQL_OPTIMIZATION))
      return;
   uint now = GetTickCount();
   if(now - g_lastDashboardTick < DASHBOARD_REFRESH_MS)
      return;
   g_lastDashboardTick = now;

   string cur = AccountInfoString(ACCOUNT_CURRENCY);
   color statusColor = UI_TEXT;
   switch(g_state)
     {
      case RX_STATE_BLOCKED:         statusColor = UI_RED;    break;
      case RX_STATE_PAIR_ACTIVE:     statusColor = UI_ACCENT; break;
      case RX_STATE_PAIR_PARTIAL:
      case RX_STATE_NO_BASKET:
      case RX_STATE_BASKET_CLOSING:  statusColor = UI_AMBER;  break;
      default:                       statusColor = UI_TEXT;   break;
     }

   string sellLeg = (s.sellLimits == 0) ? "-" : DoubleToString(s.sellLimitPrice, _Digits);
   string buyLeg  = (s.buyLimits  == 0) ? "-" : DoubleToString(s.buyLimitPrice, _Digits);
   if(s.sellLimits > 1) sellLeg += StringFormat("  (x%d, cleaning up)", s.sellLimits);
   if(s.buyLimits  > 1) buyLeg  += StringFormat("  (x%d, cleaning up)", s.buyLimits);

   string basket = "-";
   string avgTp  = "-";
   string pl     = "-";
   string costs  = "-";
   string nextRec = "-";
   color plColor = UI_TEXT;
   if(s.positions > 0)
     {
      basket = StringFormat("%s x%d   %s lot", s.side == POSITION_TYPE_BUY ? "BUY" : "SELL",
                            s.positions, DoubleToString(s.totalVolume, g_volumeDigits));
      if(s.strayPositions > 0)
         basket += StringFormat("   (+%d opposite)", s.strayPositions);
      string tpText = (g_lastTP > 0.0) ? DoubleToString(g_lastTP, _Digits)
                      : (s.uniformTP && s.commonTP > 0.0 ? DoubleToString(s.commonTP, _Digits) : "pending");
      avgTp = DoubleToString(s.vwap, _Digits) + "  >  " + tpText;
      pl = StringFormat("%+.2f %s", s.floatingProfit, cur);
      plColor = (s.floatingProfit >= 0.0) ? UI_GREEN : UI_RED;
      costs = StringFormat("commission %.2f   swap %.2f", -s.entryCommission + ExitCommissionPerLot() * s.totalVolume, s.totalSwap);
      if(InpMaxPositions > 0 && s.positions >= InpMaxPositions)
         nextRec = "limit reached";
      else if(g_nextRecoveryPrice > 0.0)
         nextRec = StringFormat("%s   (ATR %s)", DoubleToString(g_nextRecoveryPrice, _Digits), DoubleToString(g_lastATR, _Digits));
     }
   string target = StringFormat("+%.2f %s net", InpBasketProfitTarget, cur);
   if(s.positions > 0 && g_lastGrossTarget > 0.0)
      target += StringFormat("   (gross %.2f)", g_lastGrossTarget);

   UiRow(0,  "status", "Status",        g_statusText, statusColor);
   UiRow(1,  "symbol", "Symbol",        StringFormat("%s   %s account", g_symbol, cur), UI_TEXT);
   UiRow(2,  "sell",   "Sell Limit",    sellLeg, s.sellLimits > 0 ? UI_TEXT : UI_MUTED);
   UiRow(3,  "buy",    "Buy Limit",     buyLeg,  s.buyLimits  > 0 ? UI_TEXT : UI_MUTED);
   UiRow(4,  "basket", "Basket",        basket, UI_TEXT);
   UiRow(5,  "avgtp",  "Average > TP",  avgTp, UI_TEXT);
   UiRow(6,  "pl",     "Floating P/L",  pl, plColor);
   UiRow(7,  "target", "Target",        target, UI_ACCENT);
   UiRow(8,  "costs",  "Costs",         costs, UI_MUTED);
   UiRow(9,  "next",   "Next Recovery", nextRec, UI_TEXT);
   UiRow(10, "event",  "Last Event",    StringSubstr(g_lastEvent, 0, 46), UI_MUTED);
   ChartRedraw();
  }
